== in the environment ('nobf' alone for the default cache model); tools/fixtures/server.py PORT serves the pages
import gi, os, sys
gi.require_version('Gtk','4.0'); gi.require_version('WebKit','6.0')
from gi.repository import Gtk, WebKit, GLib
port = sys.argv[1]
mode = sys.argv[2] if len(sys.argv) > 2 else ''
ctx = WebKit.WebContext.get_default()
if 'docbrowser' in mode: ctx.set_cache_model(WebKit.CacheModel.DOCUMENT_BROWSER)
if 'viewer' in mode: ctx.set_cache_model(WebKit.CacheModel.DOCUMENT_VIEWER)
def count():
    me = os.getpid(); kids = {}
    for p in os.listdir('/proc'):
        if p.isdigit():
            try:
                pp = int(open(f'/proc/{p}/stat').read().rsplit(')',1)[1].split()[1]); kids.setdefault(pp,[]).append(int(p))
            except Exception: pass
    todo=[me]; n=0
    while todo:
        x=todo.pop(); todo+=kids.get(x,[])
        try:
            if open(f'/proc/{x}/cmdline').read().startswith('/usr/lib/webkitgtk-6.0/WebKitWebProcess'): n+=1
        except Exception: pass
    return n
urls = ['file://' + os.path.abspath('tools/fixtures/hello.html'), f'http://127.0.0.1:{port}/find.html', f'http://localhost:{port}/find.html', f'http://127.0.0.2:{port}/find.html', f'http://127.0.0.3:{port}/find.html']
app = Gtk.Application(application_id=f'test.mini{os.getpid()}')
def act(app):
    win = Gtk.ApplicationWindow(application=app, default_width=800, default_height=600)
    s = WebKit.Settings()
    if 'nobf' in mode:
        fl = WebKit.Settings.get_all_features()
        for i in range(fl.get_length()):
            f = fl.get(i)
            if f.get_identifier() == 'UsesBackForwardCache': s.set_feature_enabled(f, False)
    v = WebKit.WebView(settings=s); win.set_child(v); win.present()
    state = {'i': 0}
    def step():
        if state['i'] < len(urls):
            v.load_uri(urls[state['i']]); state['i'] += 1
            GLib.timeout_add(2500, lambda: (print(f'  after {urls[state["i"]-1]}: {count()} web processes', flush=True), step(), False)[2])
        else:
            state['t'] = 0
            def later():
                state['t'] += 20
                print(f'  {state["t"]} s later: {count()} web processes', flush=True)
                if state['t'] >= int(os.environ.get('WAITS', '0')): app.quit(); return False
                return True
            GLib.timeout_add(20000, later)
        return False
    step()
app.connect('activate', act); app.run([])
